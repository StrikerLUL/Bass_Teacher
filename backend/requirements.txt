# Stage 1 — stem separation.
# Pulls PyTorch (~2.5 GB with the CUDA wheels). If you want a specific CUDA
# build, install torch first so pip does not pick the CPU-only wheel:
#   pip install "torch==2.8.0" "torchaudio==2.8.0" --index-url https://download.pytorch.org/whl/cu126
#
# torch and torchaudio must always be the *same* version: torchaudio's compiled
# extension only loads against the torch it was built for (otherwise Demucs
# dies with "Could not load this library: libtorchaudio.pyd"). And they must
# stay below 2.9, where torchaudio.save needs torchcodec, which Demucs 4.0.1
# does not use.
torch>=2.1,<2.9
torchaudio>=2.1,<2.9
demucs>=4.0.1

# Stage 2 — transcription, default engine.
# CREPE tracks a single f0 per frame, which is what a separated bass stem is.
# Measured against the bass stem's own spectrum it beat Basic Pitch on harmonic
# fit, octave errors, onset recall and playability -- see compare_engines.py.
torchcrepe>=0.0.20

# Stage 2 — alternative engine (--engine basic-pitch).
# The [onnx] extra runs the model on onnxruntime instead of TensorFlow: ~600 MB
# smaller, and faster on CPU for a network this size. Use basic-pitch[tf] only
# if you specifically need the TensorFlow backend.
basic-pitch[onnx]>=0.4.0

# Used directly by processor.py for durations and --preview trimming.
# (librosa, numpy, pretty_midi and onnxruntime arrive with basic-pitch.)
soundfile>=0.12.1
